**Goal:** Prove the assembled Milk.in product works across every core journey, device, locale, and network condition — with the evidence documented and regression-gated in CI.

## Global Constraints

- **Branch:** `feat/d29-milk-qa`. PR targets `dev`, never `main`. PR title `feat(d29): milk qa`.
- **No new features.** Product code changes only to fix bugs the QA pass surfaces.
- **No `test.skip` on a failing journey.** A red test is a bug to fix or a scope decision to surface — never papered over.
- **Every new spec must be seen failing first**, for the right reason, before it is made green.

```ts
import { expect, request, type APIRequestContext, type Page } from "@playwright/test";

export const MILK = "http://localhost:3000";
export const AGRI = "http://localhost:3002";
export const STAFF_PHONE = "+919000000029"; // seed_e2e_milk.py staff identity

/** Cookie-authenticated API context for a seeded phone: the same OTP →
 * /auth/login progressive flow the browser uses, driven over HTTP. */
export async function apiAs(phone: string): Promise<APIRequestContext> {
  const bootstrap = await request.newContext({ baseURL: API });
  await resetOtpThrottle(phone);
  const requested = await bootstrap.post("/auth/otp/request", {
    data: { phone, purpose: "login" },
  });
  expect(requested.ok()).toBeTruthy();
  const code = await peekOtp(phone);
  const verify = await bootstrap.post("/auth/otp/verify", {
    data: { phone, purpose: "login", code },
  });
  expect(verify.ok()).toBeTruthy();
  const { otp_proof } = (await verify.json()) as { otp_proof: string };
  const login = await bootstrap.post("/auth/login", { data: { otp_proof } });
  expect(login.ok()).toBeTruthy();
  // agri_sid is Secure; the request-context jar won't replay it over plain
  // http://127.0.0.1, so carry it as an explicit header instead.
  const state = await bootstrap.storageState();
  const sid = state.cookies.find((c) => c.name === "agri_sid")?.value;
  expect(sid).toBeTruthy();
  await bootstrap.dispose();
  return request.newContext({
    baseURL: API,
    extraHTTPHeaders: { cookie: `agri_sid=${sid}` },
  });
}

/** Staff-role API context for the D29 moderation steps. */
export function staffApi(): Promise<APIRequestContext> {
  return apiAs(STAFF_PHONE);
}

/** Every page load races the header's AuthCluster silent-SSO probe (D10):
 * a fresh visitor bounces through /api/auth/login?silent=1 and back before
 * settling. Interacting before that resolves can hit a page mid-navigation
 * and lose client state. Wait for the logged-out "Login" button first. */
export async function waitForHeaderSettled(page: Page): Promise<void> {
  await expect(page.getByRole("button", { name: /^login$/i })).toBeVisible({ timeout: 20_000 });
}
```

```ts
import { expect, test, type Page } from "@playwright/test";

import { MILK, waitForHeaderSettled } from "./helpers";

const MIN_TAP = 44; // design-system.md — 44×44 CSS px

/** Every visible, enabled control must be tappable with a thumb. Reports ALL
 * offenders at once — fixing them one failure per run wastes the whole point. */
async function assertTapTargets(page: Page, where: string): Promise<void> {
  const offenders = await page.evaluate((min) => {
    const out: { label: string; w: number; h: number }[] = [];
    const nodes = document.querySelectorAll<HTMLElement>(
      "a[href], button, input, select, textarea, [role='button']",
    );
    for (const el of nodes) {
      if (el.hasAttribute("disabled") || el.getAttribute("aria-hidden") === "true") continue;
      const style = getComputedStyle(el);
      if (style.display === "none" || style.visibility === "hidden") continue;
      // sr-only inputs are intentionally 1px; their visible label carries the
      // target (the `peer` pattern in review-form.tsx)
      if (el.className.includes("sr-only")) continue;
      const r = el.getBoundingClientRect();
      if (r.width === 0 && r.height === 0) continue;
      if (r.width < min || r.height < min) {
        out.push({
          label: (el.getAttribute("aria-label") || el.textContent || el.tagName).trim().slice(0, 40),
          w: Math.round(r.width),
          h: Math.round(r.height),
        });
      }
    }
    return out;
  }, MIN_TAP);

  expect(
    offenders,
    `${where} has controls under ${MIN_TAP}px: ${JSON.stringify(offenders, null, 2)}`,
  ).toEqual([]);
}
```

```ts
  { name: "search", path: "/search" },

      await page.goto(`${MILK}${screen.path}`);
      await waitForHeaderSettled(page);

        .withTags(["wcag2a", "wcag2aa", "wcag21a", "wcag21aa"])

      expect(

        blocking.map((v) => `${v.id}: ${v.help} (${v.nodes.length} nodes)`).join("\n"),
      ).toEqual([]);
    });

    await expect(page.getByRole("link", { name: /login to view contact/i })).toBeVisible();
  });

  test("keyboard focus is visible on the primary CTA", async ({ page }) => {
    await page.goto(`${MILK}/coimbatore/641001`);
    await waitForHeaderSettled(page);
    await page.keyboard.press("Tab");
    const outline = await page.evaluate(() => {
      const el = document.activeElement as HTMLElement | null;
      if (!el) return null;
      const s = getComputedStyle(el);
      return { width: s.outlineWidth, style: s.outlineStyle };
    });
    expect(outline, "nothing took focus on first Tab").not.toBeNull();
    expect(
      outline!.style !== "none" && parseFloat(outline!.width) > 0,
      "focused element has no visible focus ring",
    ).toBeTruthy();
```
